
========== CollisionObject.cs ==========
using System;
using System.Collections;
using UnityEngine;

namespace CollisionLab
{
    /// <summary>
    /// Component attached to collision demonstration objects (Sphere and Cube).
    /// Handles physical collision callbacks, impact debounce, and visual reaction (color flash).
    /// </summary>
    [RequireComponent(typeof(Collider))]
    public class CollisionObject : MonoBehaviour
    {
        [Header("Object Configuration")]
        [SerializeField] private string objectName = "Object";
        [SerializeField] private bool isImpactor = false;

        [Header("Visual Feedback")]
        [SerializeField] private Color impactColor = new Color(0.0f, 0.9f, 0.46f, 1.0f); // Emerald flash
        [SerializeField] private float colorFlashDuration = 0.5f;

        private Rigidbody _rb;
        private Renderer _renderer;
        private Material _instancedMat;
        private PhysicMaterial _instancedPhysicMat;
        private Color _originalColor;
        private Vector3 _initialPosition;
        private Quaternion _initialRotation;
        private Coroutine _flashCoroutine;

        public string ObjectName => objectName;
        public bool IsImpactor => isImpactor;
        public Rigidbody Rigidbody => _rb;

        private void Awake()
        {
            _rb = GetComponent<Rigidbody>();
            _renderer = GetComponent<Renderer>();

            if (_renderer != null)
            {
                // Create instanced material to avoid modifying project asset files
                _instancedMat = _renderer.material;
                _originalColor = _instancedMat.color;
            }

            _initialPosition = transform.position;
            _initialRotation = transform.rotation;
        }

        /// <summary>
        /// Updates the Rigidbody mass safely.
        /// </summary>
        public void SetMass(float mass)
        {
            if (_rb == null) _rb = GetComponent<Rigidbody>();
            if (_rb != null)
            {
                _rb.mass = Mathf.Max(0.01f, mass);
            }
        }

        /// <summary>
        /// Safely updates restitution (bounciness) using a runtime instanced PhysicMaterial,
        /// ensuring shared project asset files are never modified.
        /// </summary>
        public void SetBounciness(float bounciness)
        {
            var col = GetComponent<Collider>();
            if (col == null) return;

            if (_instancedPhysicMat == null)
            {
                _instancedPhysicMat = new PhysicMaterial($"{name}_RuntimePhysicMat")
                {
                    dynamicFriction = 0.2f,
                    staticFriction = 0.2f,
                    bounceCombine = PhysicMaterialCombine.Maximum,
                    frictionCombine = PhysicMaterialCombine.Minimum
                };
                col.material = _instancedPhysicMat;
            }

            _instancedPhysicMat.bounciness = Mathf.Clamp01(bounciness);
        }

        private void OnCollisionEnter(Collision collision)
        {
            // Ignore collisions with the floor or boundaries for impact telemetry
            if (collision.gameObject.CompareTag("Floor") || collision.gameObject.name.Contains("Floor") || collision.gameObject.name.Contains("Wall"))
            {
                return;
            }

            // Trigger visual reaction
            TriggerImpactVisual();

            // Forward to manager for deduplicated telemetry logging
            if (CollisionLabManager.Instance != null)
            {
                CollisionLabManager.Instance.ReportCollision(this, collision);
            }
        }

        /// <summary>
        /// Flashes the object's color temporarily on collision impact.
        /// </summary>
        public void TriggerImpactVisual()
        {
            if (_renderer == null || _instancedMat == null) return;

            if (_flashCoroutine != null)
            {
                StopCoroutine(_flashCoroutine);
            }
            _flashCoroutine = StartCoroutine(FlashRoutine());
        }

        private IEnumerator FlashRoutine()
        {
            _instancedMat.color = impactColor;
            
            // Set emission if standard shader supports it
            if (_instancedMat.HasProperty("_EmissionColor"))
            {
                _instancedMat.EnableKeyword("_EMISSION");
                _instancedMat.SetColor("_EmissionColor", impactColor * 0.6f);
            }

            var elapsed = 0f;
            while (elapsed < colorFlashDuration)
            {
                elapsed += Time.deltaTime;
                var t = elapsed / colorFlashDuration;
                // Smoothly lerp back to original color
                _instancedMat.color = Color.Lerp(impactColor, _originalColor, t);
                if (_instancedMat.HasProperty("_EmissionColor"))
                {
                    _instancedMat.SetColor("_EmissionColor", Color.Lerp(impactColor * 0.6f, Color.black, t));
                }
                yield return null;
            }

            _instancedMat.color = _originalColor;
            if (_instancedMat.HasProperty("_EmissionColor"))
            {
                _instancedMat.SetColor("_EmissionColor", Color.black);
            }
            _flashCoroutine = null;
        }

        /// <summary>
        /// Resets position, rotation, velocities, and visual material to the initial state.
        /// </summary>
        public void ResetToInitialState()
        {
            if (_flashCoroutine != null)
            {
                StopCoroutine(_flashCoroutine);
                _flashCoroutine = null;
            }

            transform.position = _initialPosition;
            transform.rotation = _initialRotation;

            if (_rb != null)
            {
                _rb.velocity = Vector3.zero;
                _rb.angularVelocity = Vector3.zero;
                _rb.Sleep();
                _rb.WakeUp();
            }

            if (_instancedMat != null)
            {
                _instancedMat.color = _originalColor;
                if (_instancedMat.HasProperty("_EmissionColor"))
                {
                    _instancedMat.SetColor("_EmissionColor", Color.black);
                }
            }
        }

        private void OnDestroy()
        {
            if (_instancedMat != null) Destroy(_instancedMat);
            if (_instancedPhysicMat != null) Destroy(_instancedPhysicMat);
        }
    }
}

    [MenuItem("VR/Build CollisionLab_VR Scene")]
    public static void BuildVRScene()
    {
        Debug.Log("[VRSceneBuilder] Starting VR Scene Construction...");

        // 1. Open CollisionLab.unity as base
        string sourceScenePath = "Assets/Scenes/CollisionLab.unity";
        string targetScenePath = "Assets/Scenes/CollisionLab_VR.unity";

        var scene = EditorSceneManager.OpenScene(sourceScenePath, OpenSceneMode.Single);
        if (!scene.IsValid())
        {
            Debug.LogError($"[VRSceneBuilder] Failed to open base scene: {sourceScenePath}");
            return;
        }

        // 2. Remove desktop Main Camera and desktop Canvas & EventSystem
        var desktopCam = GameObject.Find("Main Camera");
        if (desktopCam != null)
        {
            Debug.Log("[VRSceneBuilder] Removing desktop camera...");
            UnityEngine.Object.DestroyImmediate(desktopCam);
        }

        var desktopCanvas = GameObject.Find("Canvas");
        if (desktopCanvas != null)
        {
            Debug.Log("[VRSceneBuilder] Removing desktop canvas...");
            UnityEngine.Object.DestroyImmediate(desktopCanvas);
        }

        var desktopEventSystem = GameObject.Find("EventSystem");
        if (desktopEventSystem != null)
        {
            Debug.Log("[VRSceneBuilder] Removing desktop EventSystem...");
            UnityEngine.Object.DestroyImmediate(desktopEventSystem);
        }

        // 3. Configure Floor with TeleportationArea
        var floor = GameObject.Find("Floor");
        if (floor != null)
        {
            if (floor.GetComponent<TeleportationArea>() == null)
            {
                Debug.Log("[VRSceneBuilder] Adding TeleportationArea to Floor...");
                floor.AddComponent<TeleportationArea>();
            }
        }

        // 4. Instantiate XR Interaction Setup prefab
        string xrSetupPrefabPath = "Assets/Samples/XR Interaction Toolkit/2.6.3/Starter Assets/Prefabs/XR Interaction Setup.prefab";
        var xrSetupPrefab = AssetDatabase.LoadAssetAtPath<GameObject>(xrSetupPrefabPath);
        if (xrSetupPrefab == null)
        {
            Debug.LogError($"[VRSceneBuilder] Failed to load XR Interaction Setup prefab at {xrSetupPrefabPath}");
            return;
        }

        var xrSetupInstance = (GameObject)PrefabUtility.InstantiatePrefab(xrSetupPrefab);
        xrSetupInstance.name = "XR Interaction Setup";
        xrSetupInstance.transform.position = new Vector3(0f, 0f, -4.5f);
        xrSetupInstance.transform.rotation = Quaternion.identity;
        Debug.Log("[VRSceneBuilder] Instantiated XR Interaction Setup at (0, 0, -4.5)");

        // 5. Instantiate XR Device Simulator prefab
        string simPrefabPath = "Assets/Samples/XR Interaction Toolkit/2.6.3/XR Device Simulator/XR Device Simulator.prefab";
        var simPrefab = AssetDatabase.LoadAssetAtPath<GameObject>(simPrefabPath);
        if (simPrefab != null)
        {
            var simInstance = (GameObject)PrefabUtility.InstantiatePrefab(simPrefab);
            simInstance.name = "XR Device Simulator";
            Debug.Log("[VRSceneBuilder] Instantiated XR Device Simulator.");
        }
        else
        {
            Debug.LogWarning($"[VRSceneBuilder] XR Device Simulator prefab not found at {simPrefabPath}");
        }

        // 6. Create Grabbable Test Pedestal and Cube
        CreateGrabbableTestObject();

        // 7. Create World-Space Control Canvas & UI
        var vrCanvasGO = CreateWorldSpaceCanvas(out var uiRefs);

        // 8. Add XRSceneBootstrap
        var bootstrapGO = new GameObject("XRSceneBootstrap");
        bootstrapGO.AddComponent<XRSceneBootstrap>();

        // 9. Configure CollisionLabManager
        WireCollisionLabManager(uiRefs);

        // 10. Save as CollisionLab_VR.unity
        bool saved = EditorSceneManager.SaveScene(scene, targetScenePath);
        if (saved)
        {
            Debug.Log($"[VRSceneBuilder] Successfully saved VR scene to {targetScenePath}");
        }
        else
        {
            Debug.LogError($"[VRSceneBuilder] Failed to save VR scene to {targetScenePath}");
            return;
        }

        // 11. Update EditorBuildSettings
        UpdateBuildSettings(targetScenePath);

        Debug.Log("[VRSceneBuilder] Build completed successfully!");
    }

    private static void CreateGrabbableTestObject()
    {
        // Pedestal
        var pedestal = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        pedestal.name = "CalibrationPedestal";
        pedestal.transform.position = new Vector3(1.6f, 0.4f, -3.2f);
        pedestal.transform.localScale = new Vector3(0.5f, 0.4f, 0.5f);
        var pedRenderer = pedestal.GetComponent<Renderer>();
        if (pedRenderer != null)
        {
            pedRenderer.material.color = new Color(0.18f, 0.22f, 0.28f); // Dark Slate
        }

        // Grabbable Cube
        var cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
        cube.name = "VRCalibrationCube";
        cube.transform.position = new Vector3(1.6f, 0.95f, -3.2f);
        cube.transform.localScale = new Vector3(0.25f, 0.25f, 0.25f);

        var rb = cube.GetComponent<Rigidbody>();
        if (rb == null) rb = cube.AddComponent<Rigidbody>();
        rb.mass = 0.5f;
        rb.collisionDetectionMode = CollisionDetectionMode.Continuous;

        var grab = cube.AddComponent<XRGrabInteractable>();
        grab.movementType = XRBaseInteractable.MovementType.VelocityTracking;
        grab.throwOnDetach = true;

        var cubeRenderer = cube.GetComponent<Renderer>();
        if (cubeRenderer != null)
        {
            cubeRenderer.material.color = new Color(0.95f, 0.7f, 0.1f); // Vibrant Gold
        }

        Debug.Log("[VRSceneBuilder] Created Grabbable Calibration Cube on Pedestal.");
    }

    private static void UpdateBuildSettings(string vrScenePath)
    {
        var existingScenes = EditorBuildSettings.scenes;
        var sceneList = new List<EditorBuildSettingsScene>();

        bool hasMain = false;
        bool hasCollisionDesktop = false;
        bool hasVR = false;

        foreach (var s in existingScenes)
        {
            if (s.path == "Assets/Scenes/MainScene.unity")
            {
                hasMain = true;
                sceneList.Add(s);
            }
            else if (s.path == "Assets/Scenes/CollisionLab.unity")
            {
                hasCollisionDesktop = true;
                sceneList.Add(s);
            }
            else if (s.path == vrScenePath)
            {
                hasVR = true;
            }
        }

        if (!hasMain)
        {
            sceneList.Insert(0, new EditorBuildSettingsScene("Assets/Scenes/MainScene.unity", true));
        }

        if (!hasCollisionDesktop)
        {
            sceneList.Add(new EditorBuildSettingsScene("Assets/Scenes/CollisionLab.unity", true));
        }

        sceneList.Add(new EditorBuildSettingsScene(vrScenePath, true));

        EditorBuildSettings.scenes = sceneList.ToArray();

        Debug.Log("[VRSceneBuilder] Updated EditorBuildSettings. Total scenes: " + EditorBuildSettings.scenes.Length);
        for (int i = 0; i < EditorBuildSettings.scenes.Length; i++)
        {
            Debug.Log($"  [{i}] {EditorBuildSettings.scenes[i].path} (Enabled: {EditorBuildSettings.scenes[i].enabled})");
        }
    }

        /// <summary>
        /// Launches the sphere with a direct linear velocity impulse.
        /// This ensures the initial launch speed is strictly independent of the sphere's mass.
        /// </summary>
        public void LaunchSphere()
        {
            if (sphereRigidbody == null) return;

            ApplyPhysicsConfiguration();

            // Clear any lingering velocities
            sphereRigidbody.velocity = Vector3.zero;
            sphereRigidbody.angularVelocity = Vector3.zero;
            if (cubeRigidbody != null)
            {
                cubeRigidbody.velocity = Vector3.zero;
                cubeRigidbody.angularVelocity = Vector3.zero;
            }

            if (contactVisualizer != null) contactVisualizer.Clear();

            // Direct linear velocity assignment guarantees mass-independence
            var dir = launchDirection.normalized;
            sphereRigidbody.velocity = dir * launchVelocity;

            _lastFixedSphereVel = sphereRigidbody.velocity;
            _lastFixedCubeVel = Vector3.zero;
            _preImpactSphereVel = _lastFixedSphereVel;
            _preImpactCubeVel = Vector3.zero;

            _currentState = SimulationState.InMotion;
            UpdateTelemetryState();
        }

        /// <summary>
        /// Callback triggered by CollisionObject on physical contact.
        /// Deduplicates multi-body callbacks using pair-hash and physics step deduplication,
        /// avoiding artificial cooldowns while guaranteeing exactly one count per physical collision.
        /// </summary>
        public void ReportCollision(CollisionObject source, Collision collision)
        {
            if (source == null || collision == null || collision.gameObject == null) return;

            int idA = source.gameObject.GetInstanceID();
            int idB = collision.gameObject.GetInstanceID();

            int minId = idA < idB ? idA : idB;
            int maxId = idA < idB ? idB : idA;
            int pairKey = unchecked((minId * 397) ^ maxId);

            int currentStep = Time.frameCount;
            if (_lastPhysicsTickByPair.TryGetValue(pairKey, out int lastStep) && lastStep == currentStep)
            {
                return;
            }

            _lastPhysicsTickByPair[pairKey] = currentStep;
            _impactCount++;
            _currentState = SimulationState.Collided;

            // Capture pre-impact velocities from the physics step before impact
            var v1i = _lastFixedSphereVel.sqrMagnitude > 0.01f ? _lastFixedSphereVel : launchDirection.normalized * launchVelocity;
            var v2i = _lastFixedCubeVel;

            // Extract contact manifold
            var contactPoint = Vector3.zero;
            var contactNormal = Vector3.zero;
            if (collision.contactCount > 0)
            {
                var contact = collision.GetContact(0);
                contactPoint = contact.point;
                contactNormal = contact.normal;
            }

            // Extract PhysX relative velocity & impulse
            var relVel = collision.relativeVelocity.magnitude;
            var impulse = collision.impulse.magnitude;

            // Visual contact marker & normal line
            if (contactVisualizer != null)
            {
                contactVisualizer.ShowContact(contactPoint, contactNormal);
            }

            // Sample post-impact velocities across the collision resolution
            StartCoroutine(CapturePostCollisionData(v1i, v2i, relVel, impulse, contactPoint, contactNormal));
        }

        private IEnumerator CapturePostCollisionData(Vector3 v1i, Vector3 v2i, float relVel, float impulse, Vector3 contactPoint, Vector3 contactNormal)
        {
            // Wait for FixedUpdate to ensure PhysX collision solver has fully integrated post-impact velocities
            yield return new WaitForFixedUpdate();

            var v1f = (sphereRigidbody != null) ? sphereRigidbody.velocity : Vector3.zero;
            var v2f = (cubeRigidbody != null) ? cubeRigidbody.velocity : Vector3.zero;

            // 1. Vector Momentum Conservation Analysis
            // p = m1*v1 + m2*v2
            var pInitial = sphereMass * v1i + cubeMass * v2i;
            var pFinal = sphereMass * v1f + cubeMass * v2f;
            var pDelta = pFinal - pInitial;

            float pInitMag = pInitial.magnitude;
            float pErrorPercent = (pInitMag > 0.001f) ? (pDelta.magnitude / pInitMag) * 100f : 0f;

            // 2. Kinetic Energy Analysis
            // KE = 0.5 * m * v^2
            float keInitial = 0.5f * sphereMass * v1i.sqrMagnitude + 0.5f * cubeMass * v2i.sqrMagnitude;
            float keFinal = 0.5f * sphereMass * v1f.sqrMagnitude + 0.5f * cubeMass * v2f.sqrMagnitude;
            float deltaKE = keFinal - keInitial;
            float keRetainedPercent = (keInitial > 0.001f) ? (keFinal / keInitial) * 100f : 0f;

            // 3. Coefficient of Restitution along Contact Normal
            // e = - (v1f - v2f) · n / ((v1i - v2i) · n)
            var norm = contactNormal.normalized;
            if (norm.sqrMagnitude < 0.001f) norm = Vector3.right;

            // Approach relative speed along normal
            float approachSpeed = Vector3.Dot(v1i - v2i, -norm);
            if (approachSpeed < 0.001f) approachSpeed = Vector3.Dot(v1i - v2i, norm);
            approachSpeed = Mathf.Abs(approachSpeed);

            // Separation relative speed along normal
            float separationSpeed = Vector3.Dot(v2f - v1f, -norm);
            if (separationSpeed < 0.001f) separationSpeed = Vector3.Dot(v2f - v1f, norm);
            separationSpeed = Mathf.Abs(separationSpeed);

            float measuredRestitution = (approachSpeed > 0.05f) ? (separationSpeed / approachSpeed) : restitution;
            measuredRestitution = Mathf.Clamp(measuredRestitution, 0f, 1.2f);

            // Update UI telemetry displays
            UpdateTelemetryCollision(relVel, impulse, contactPoint, contactNormal, pInitial, pFinal, pErrorPercent, keInitial, keFinal, deltaKE, keRetainedPercent, measuredRestitution);
        }

        /// <summary>
        /// Resets positions, rotations, linear/angular velocities, colors, and telemetry.
        /// Preserves the user-selected slider configuration values.
        /// </summary>
        public void ResetSimulation()
        {
            _currentState = SimulationState.Ready;
            _impactCount = 0;
            _lastPhysicsTickByPair.Clear();

            _lastFixedSphereVel = Vector3.zero;
            _lastFixedCubeVel = Vector3.zero;
            _preImpactSphereVel = Vector3.zero;
            _preImpactCubeVel = Vector3.zero;

            if (sphereObject != null) sphereObject.ResetToInitialState();
            if (cubeObject != null) cubeObject.ResetToInitialState();

            if (sphereRigidbody != null)
            {
                sphereRigidbody.velocity = Vector3.zero;
                sphereRigidbody.angularVelocity = Vector3.zero;
            }

            if (cubeRigidbody != null)
            {
                cubeRigidbody.velocity = Vector3.zero;
                cubeRigidbody.angularVelocity = Vector3.zero;
            }

            if (sphereVectorVisualizer != null) sphereVectorVisualizer.ClearVector();
            if (cubeVectorVisualizer != null) cubeVectorVisualizer.ClearVector();
            if (contactVisualizer != null) contactVisualizer.Clear();

            // Reapply current slider configuration
            ApplyPhysicsConfiguration();

            UpdateTelemetryReset();
        }
